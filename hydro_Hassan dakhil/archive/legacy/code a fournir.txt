interface HydroDataContextType {
  stations: Station[];
  catchments: Catchment[];
  timeseries: Timeseries[];
  reservoirs: Reservoir[];
  loading: boolean;
  error: string | null;
}

const HydroDataContext = createContext<HydroDataContextType>({
  stations: [],
  catchments: [],
  timeseries: [],
  reservoirs: [],
  loading: true,
  error: null,
});

export const HydroDataProvider = ({ children }: { children: ReactNode }) => {
  const [stations, setStations] = useState<Station[]>([]);
  const [catchments, setCatchments] = useState<Catchment[]>([]);
  const [timeseries, setTimeseries] = useState<Timeseries[]>([]);
  const [reservoirs, setReservoirs] = useState<Reservoir[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function fetchData() {
      try {
        const { stations, catchments, timeseries, reservoirs } = await getDashboardData();
        setStations(stations);
        setCatchments(catchments);
        setTimeseries(timeseries);
        setReservoirs(reservoirs);
      } catch (err) {
        setError(err instanceof Error ? err.message : 'Erreur inconnue');
      } finally {
        setLoading(false);
      }
    }

    fetchData();
  }, []);

  return (
    <HydroDataContext.Provider value={{ stations, catchments, timeseries, reservoirs, loading, error }}>
      {children}
    </HydroDataContext.Provider>
  );
};

import { apiGet, qs } from "./client";
import type {
  Station,
  Catchment,
  Timeseries,
  Reservoir,
  ModelRun,
  Landcover,
  LandcoverSummaryItem,
  SpatialFeaturesResponse,
  DashboardStats,
  HealthResponse,
  FilterOptions,
} from "../types/hydro";

export const hydroApi = {
  health: () => apiGet<HealthResponse>("/hydro/health"),

  getStations: (filter?: FilterOptions) => {
    // ton backend accepte stationIds en JSON string + limit (selon ton code actuel)
    const query = qs({
      limit: filter?.limit,
      stationIds: filter?.stationIds?.length
        ? JSON.stringify(filter.stationIds)
        : undefined,
    });
    return apiGet<Station[]>(`/hydro/stations${query}`);
  },

  getCatchments: (filter?: FilterOptions) => {
    const query = qs({
      catchmentIds: filter?.catchmentIds?.length
        ? JSON.stringify(filter.catchmentIds)
        : undefined,
    });
    return apiGet<Catchment[]>(`/hydro/catchments${query}`);
  },

  getTimeseries: (stationId?: number, propertyId?: number, runId?: number) => {
    const query = qs({ stationId, propertyId, runId });
    return apiGet<Timeseries[]>(`/hydro/timeseries${query}`);
  },

  getReservoirs: () => apiGet<Reservoir[]>("/hydro/reservoirs"),

  getModelRuns: (isObserved?: boolean) => {
    const query = qs({ isObserved });
    return apiGet<ModelRun[]>(`/hydro/model-runs${query}`);
  },

  getLandcover: () => apiGet<Landcover[]>("/hydro/landcover"),

  getLandcoverSummary: (catchmentId: number, periodId?: number) => {
    const query = qs({ periodId });
    return apiGet<LandcoverSummaryItem[]>(
      `/hydro/catchments/${catchmentId}/landcover-summary${query}`
    );
  },

  getFeaturesInBounds: (b: {
    minLng: number;
    minLat: number;
    maxLng: number;
    maxLat: number;
  }) => {
    const query = qs(b);
    return apiGet<SpatialFeaturesResponse>(`/hydro/spatial/features${query}`);
  },

  getDashboardStats: () => apiGet<DashboardStats>("/hydro/stats"),
};

// Option pratique pour ton HydroDataContext (tu l’utilises déjà) :contentReference[oaicite:6]{index=6}
export async function getDashboardData() {
  const [stations, catchments, timeseries, reservoirs] = await Promise.all([
    hydroApi.getStations({ limit: 50 }),
    hydroApi.getCatchments(),
    hydroApi.getTimeseries(),
    hydroApi.getReservoirs(),
  ]);
  return { stations, catchments, timeseries, reservoirs };
}

import { useState, useEffect, useCallback } from 'react';
import { hydroApi } from '@/api/hydro';

type Status = 'idle' | 'loading' | 'success' | 'error';

interface UseFetchOptions<T> {
  initialData?: T;
  enabled?: boolean;
  refetchInterval?: number;
  onSuccess?: (data: T) => void;
  onError?: (error: Error) => void;
}

interface UseFetchResult<T> {
  data: T | null;
  error: Error | null;
  status: Status;
  isLoading: boolean;
  isError: boolean;
  isSuccess: boolean;
  refetch: () => Promise<void>;
}

export function useFetch<T>(
  fetchFn: () => Promise<T>,
  options: UseFetchOptions<T> = {}
): UseFetchResult<T> {
  const {
    initialData = null,
    enabled = true,
    refetchInterval,
    onSuccess,
    onError,
  } = options;

  const [data, setData] = useState<T | null>(initialData);
  const [error, setError] = useState<Error | null>(null);
  const [status, setStatus] = useState<Status>('idle');

  const isLoading = status === 'loading';
  const isError = status === 'error';
  const isSuccess = status === 'success';

  const fetchData = useCallback(async () => {
    if (!enabled) return;

    setStatus('loading');
    setError(null);

    try {
      const result = await fetchFn();
      setData(result);
      setStatus('success');
      onSuccess?.(result);
    } catch (err) {
      const error = err instanceof Error ? err : new Error('Unknown error');
      setError(error);
      setStatus('error');
      onError?.(error);
    }
  }, [fetchFn, enabled, onSuccess, onError]);

  useEffect(() => {
    if (enabled) {
      fetchData();
    }
  }, [enabled, fetchData]);

  useEffect(() => {
    if (!refetchInterval) return;

    const intervalId = setInterval(() => {
      if (enabled) {
        fetchData();
      }
    }, refetchInterval);

    return () => clearInterval(intervalId);
  }, [refetchInterval, enabled, fetchData]);

  return {
    data,
    error,
    status,
    isLoading,
    isError,
    isSuccess,
    refetch: fetchData,
  };
}

// Hook simple pour les données hydrologiques
export function useHydroData<T>(
  apiFunction: () => Promise<T>,
  options?: UseFetchOptions<T>
) {
  return useFetch(apiFunction, options);
}

};

export const useTimeseries = (params?: { stationId?: string; variable?: string }, options?: UseFetchOptions<Array<{ timestamp: string; value: number }>>) => {
  return useFetch(() => hydroApi.getTimeseries(params || {}), options);

interface ChartDataPoint {
  date: string;
  [key: string]: string | number;
}

// Définir un type pour les mesures
interface ChartMeasurement {
  date?: string;
  timestamp?: string;
  time?: string;
  value: number;
  [key: string]: unknown;
}

export function TimeSeriesChart({ filters }: TimeSeriesChartProps) {
  const chartData = useMemo(() => {
    if (filters.variables.length === 0)
      return { data: [] as ChartDataPoint[], stats: [], selectedVars: [] };

    const selectedVars = variables.filter((v) =>
      filters.variables.includes(v.id)
    );
    const dataMap = new Map<string, ChartDataPoint>();
    const stats: Array<{
      variable: string;
      unit: string;
      min: number;
      max: number;
      mean: number;
    }> = [];

    selectedVars.forEach((variable) => {
      const generator = mockDataGenerators[variable.code];
      if (!generator) return;

      const measurements = generateMockTimeSeries(
        filters.startDate,
        filters.endDate,
        variable.code,
        generator.base,
        generator.amplitude,
        generator.noise
      );

      const varStats = calculateStats(measurements);
      stats.push({
        variable: variable.name,
        unit: variable.unit,
        min: Math.round(varStats.min * 100) / 100,
        max: Math.round(varStats.max * 100) / 100,
        mean: Math.round(varStats.mean * 100) / 100,
      });

      measurements.forEach((m) => {
        // Convertir la mesure en type connu
        const measurement = m as unknown as ChartMeasurement;

        // Trouver la clé de date - essayer différentes propriétés
        let dateKey = "";
        if (typeof measurement === "object" && measurement !== null) {
          if ("date" in measurement && typeof measurement.date === "string") {
            dateKey = measurement.date;
          } else if (
            "timestamp" in measurement &&
            typeof measurement.timestamp === "string"
          ) {
            dateKey = measurement.timestamp;
          } else if (
            "time" in measurement &&
            typeof measurement.time === "string"
          ) {
            dateKey = measurement.time;
          } else if ("value" in measurement) {
            dateKey = String(measurement.value);
          }
        }

        if (dateKey) {
          const existing = dataMap.get(dateKey) || { date: dateKey };
          existing[variable.code] = measurement.value;
          dataMap.set(dateKey, existing);
        }
      });
    });

    // Sample data for better performance (max 365 points)
    const allData = Array.from(dataMap.values()).sort(
      (a, b) => new Date(a.date).getTime() - new Date(b.date).getTime()
    );

    const step = Math.ceil(allData.length / 365);
    const sampledData = allData.filter((_, i) => i % step === 0);

    return { data: sampledData, stats, selectedVars };
  }, [filters]);

  const exportCSV = () => {
    if (chartData.data.length === 0) return;

    const headers = [
      "Date",
      ...chartData.selectedVars.map((v) => `${v.name} (${v.unit})`),
    ];

      d.date,
      ...chartData.selectedVars.map((v) => d[v.code] ?? ""),
    ]);

    const csv = [headers, ...rows].map((r) => r.join(",")).join("\n");
    const blob = new Blob([csv], { type: "text/csv" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `hydro_data_${filters.startDate}_${filters.endDate}.csv`;
    a.click();
  };

  if (filters.variables.length === 0) {
    return (
      <div className="hydro-card h-full flex items-center justify-center">
        <div className="text-center text-muted-foreground">
          <TrendingUp className="w-12 h-12 mx-auto mb-3 opacity-50" />
          <p>Sélectionnez des variables pour afficher les graphiques</p>
        </div>
      </div>
    );
  }

  return (
    <div className="hydro-card">
      <div className="hydro-card-header">
        <h3 className="font-semibold">Séries Temporelles</h3>
        <Button variant="outline" size="sm" onClick={exportCSV}>
          <Download className="w-4 h-4 mr-2" />
          Export CSV
        </Button>
      </div>

      <div className="hydro-card-body">
        {/* Stats Cards */}
        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-6 gap-3 mb-6">
          {chartData.stats.map((stat, i) => (
            <div key={stat.variable} className="stat-card">
              <div
                className="text-xs text-muted-foreground mb-1 truncate"
                title={stat.variable}
              >
                {stat.variable}
              </div>
              <div className="space-y-1">
                <div className="flex items-center justify-between text-xs">
                  <span className="text-muted-foreground">Min</span>
                  <span className="font-mono font-medium text-blue-600">
                    {stat.min} {stat.unit}
                  </span>
                </div>
                <div className="flex items-center justify-between text-xs">
                  <span className="text-muted-foreground">Max</span>
                  <span className="font-mono font-medium text-red-600">
                    {stat.max} {stat.unit}

                <div className="flex items-center justify-between text-xs">

                  <span className="font-mono font-medium text-foreground">
                    {stat.mean} {stat.unit}
                  </span>
                </div>
              </div>
            </div>
          ))}
        </div>

        {/* Chart */}
        <div className="chart-container">

interface DataTableProps {

const ROWS_PER_PAGE = 15;

// Définir le type de mesure basé sur ce que retourne generateMockTimeSeries
interface TableMeasurement {
  date?: string;
  timestamp?: string;
  time?: string;
  value: number;
  [key: string]: unknown; // Signature d'index pour flexibilité
}

export function DataTable({ filters }: DataTableProps) {
  const [sortColumn, setSortColumn] = useState<string>("date");
  const [sortDirection, setSortDirection] = useState<"asc" | "desc">("desc");
  const [searchTerm, setSearchTerm] = useState("");
  const [currentPage, setCurrentPage] = useState(1);

  const tableData = useMemo(() => {
    if (filters.variables.length === 0) return { rows: [], columns: [] };

    const selectedVars = variables.filter((v) =>
      filters.variables.includes(v.id)
    );
    const columns = [
      { key: "date", label: "Date" },
      ...selectedVars.map((v) => ({
        key: v.code,
        label: `${v.name} (${v.unit})`,
      })),
    ];

    const dataMap = new Map<string, Record<string, string | number>>();

    selectedVars.forEach((variable) => {
      const generator = mockDataGenerators[variable.code];
      if (!generator) return;

      const measurements = generateMockTimeSeries(
        filters.startDate,
        filters.endDate,
        variable.code,
        generator.base,
        generator.amplitude,
        generator.noise
      );

      measurements.forEach((m) => {
        // Convertir en unknown puis en TableMeasurement pour éviter les erreurs de type
        const measurement = m as unknown as TableMeasurement;

        // Extraire la date - essayer différentes propriétés possibles
        let dateKey = "";
        if (typeof measurement === "object" && measurement !== null) {
          if ("date" in measurement && typeof measurement.date === "string") {
            dateKey = measurement.date;
          } else if (
            "timestamp" in measurement &&
            typeof measurement.timestamp === "string"
          ) {
            dateKey = measurement.timestamp;
          } else if (
            "time" in measurement &&
            typeof measurement.time === "string"
          ) {
            dateKey = measurement.time;
          } else if ("value" in measurement) {
            dateKey = String(measurement.value);
          }
        }

        if (dateKey) {
          const existing = dataMap.get(dateKey) || { date: dateKey };
          existing[variable.code] = measurement.value;
          dataMap.set(dateKey, existing);
        }
      });
    });

    const rows = Array.from(dataMap.values());
    return { rows, columns };
  }, [filters]);

  const sortedAndFilteredData = useMemo(() => {
    let data = [...tableData.rows];

    // Filter
    if (searchTerm) {
      data = data.filter((row) =>
        Object.values(row).some((val) =>
          String(val).toLowerCase().includes(searchTerm.toLowerCase())
        )
      );
    }

    // Sort
    data.sort((a, b) => {
      const aVal = a[sortColumn];
      const bVal = b[sortColumn];

      if (sortColumn === "date") {
        const dateA = new Date(aVal as string).getTime();
        const dateB = new Date(bVal as string).getTime();
        return sortDirection === "asc" ? dateA - dateB : dateB - dateA;
      }

      const numA = Number(aVal) || 0;
      const numB = Number(bVal) || 0;
      return sortDirection === "asc" ? numA - numB : numB - numA;
    });

    return data;
  }, [tableData.rows, sortColumn, sortDirection, searchTerm]);

  const paginatedData = useMemo(() => {
    const start = (currentPage - 1) * ROWS_PER_PAGE;
    return sortedAndFilteredData.slice(start, start + ROWS_PER_PAGE);
  }, [sortedAndFilteredData, currentPage]);

  const totalPages = Math.ceil(sortedAndFilteredData.length / ROWS_PER_PAGE);

  const handleSort = (column: string) => {
    if (sortColumn === column) {
      setSortDirection((prev) => (prev === "asc" ? "desc" : "asc"));
    } else {
      setSortColumn(column);
      setSortDirection("desc");
    }
  };

  const exportCSV = () => {
    const headers = tableData.columns.map((c) => c.label);
    const rows = sortedAndFilteredData.map((row) =>
      tableData.columns.map((c) => row[c.key] ?? "")
    );

    const csv = [headers, ...rows].map((r) => r.join(",")).join("\n");

    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `hydro_table_${filters.startDate}_${filters.endDate}.csv`;
    a.click();
  };

  if (filters.variables.length === 0) {
    return (
      <div className="hydro-card h-full flex items-center justify-center">
        <div className="text-center text-muted-foreground">
          <p>Sélectionnez des variables pour afficher les données</p>
        </div>
      </div>
    );

  return (
    <div className="hydro-card">
      <div className="hydro-card-header">
        <h3 className="font-semibold">Données Tabulaires</h3>
        <div className="flex items-center gap-3">
          <div className="relative">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />

            <Download className="w-4 h-4 mr-2" />
            Export CSV
          </Button>
        </div>
      </div>

      <div className="overflow-x-auto">
        <table className="data-table">
          <thead>
            <tr>
              {tableData.columns.map((column) => (
                <th
                  key={column.key}
                  onClick={() => handleSort(column.key)}
                  className="cursor-pointer hover:bg-muted/70 transition-colors"
                >
                  <div className="flex items-center gap-2">
                    <span className="truncate">{column.label}</span>
                    <div className="flex flex-col">
                      <ChevronUp
                        className={cn(
                          "w-3 h-3 -mb-1",
                          sortColumn === column.key && sortDirection === "asc"
                            ? "text-primary"
                            : "text-muted-foreground/40"
                        )}
                      />
                      <ChevronDown
                        className={cn(
                          "w-3 h-3 -mt-1",
                          sortColumn === column.key && sortDirection === "desc"
                            ? "text-primary"
                            : "text-muted-foreground/40"
                        )}
                      />
                    </div>
                  </div>
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {paginatedData.map((row, i) => (
              <tr key={i}>
                {tableData.columns.map((column) => (
                  <td key={column.key} className="font-mono text-sm">
                    {column.key === "date"
                      ? new Date(row[column.key] as string).toLocaleDateString(
                          "fr-FR"
                        )
                      : typeof row[column.key] === "number"
                      ? (row[column.key] as number).toFixed(2)
                      : row[column.key] ?? "-"}
                  </td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {/* Pagination */}
      <div className="px-5 py-4 border-t border-border flex items-center justify-between">
        <div className="text-sm text-muted-foreground">
          {sortedAndFilteredData.length} enregistrements • Page {currentPage} /{" "}
          {totalPages}
        </div>
        <div className="flex items-center gap-2">
          <Button
            variant="outline"
            size="sm"
            onClick={() => setCurrentPage((p) => Math.max(1, p - 1))}
            disabled={currentPage === 1}
          >
            <ChevronLeft className="w-4 h-4" />
          </Button>
          <Button
            variant="outline"
            size="sm"
            onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))}
            disabled={currentPage === totalPages}
          >
            <ChevronRight className="w-4 h-4" />
          </Button>
        </div>
      </div>
    </div>
  );
}
