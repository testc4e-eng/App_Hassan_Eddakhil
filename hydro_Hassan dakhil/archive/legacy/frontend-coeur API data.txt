const API_BASE_URL = import.meta.env.VITE_API_URL || 'http://localhost:5000/api/v1/hydro';

class ApiError extends Error {
  constructor(message: string, public status?: number) {
    super(message);
    this.name = 'ApiError';
  }
}

// Utilitaire fetch avec gestion d'erreur améliorée
async function fetchAPI<T>(endpoint: string, options?: RequestInit): Promise<T> {
  const url = `${API_BASE_URL}${endpoint}`;
  
  try {
    const response = await fetch(url, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        ...options?.headers,
      },
    });

    if (!response.ok) {
      let errorMessage = `HTTP ${response.status}`;
      try {
        const errorData = await response.json();
        errorMessage = errorData.error || errorMessage;
      } catch {
        // Ignorer si pas de JSON
      }
      throw new ApiError(errorMessage, response.status);
    }

    const data: ApiResponse<T> = await response.json();
    
    if (!data.success) {
      throw new ApiError(data.error || 'API request failed');
    }

    return data.data;
  } catch (error) {
    console.error(`API Error ${endpoint}:`, error);
    if (error instanceof ApiError) {
      throw error;
    }
    throw new ApiError(error instanceof Error ? error.message : 'Network error');
  }
}

// ================ STATIONS ================
export async function getStations(filter?: FilterOptions): Promise<Station[]> {
  const params = new URLSearchParams();
  
  if (filter?.stationIds?.length) {
    params.append('stationIds', JSON.stringify(filter.stationIds));
  }
  if (filter?.limit) {
    params.append('limit', filter.limit.toString());
  }

  const query = params.toString() ? `?${params.toString()}` : '';
  return fetchAPI<Station[]>(`/stations${query}`);
}

export async function getStationById(id: number): Promise<Station> {
  return fetchAPI<Station>(`/stations/${id}`);
}

// Convertir les timeseries en format compatible avec l'ancien dashboard
export function formatTimeseriesForUI(timeseries: Timeseries[]): UIVariable[] {
  return timeseries.map(ts => ({
    id: ts.property_code || `prop-${ts.property_id}`,
    name: ts.property_name || 'Inconnu',
    unit: ts.unit || 'N/A',
    category: getVariableCategory(ts.property_code || '')
  }));
}

function getVariableCategory(code: string): string {
  if (code.includes('temp') || code.includes('precip') || code.includes('et')) {
    return 'climate';
  } else if (code.includes('discharge') || code.includes('level') || code.includes('volume')) {
    return 'hydraulic';
  } else if (code.includes('sediment') || code.includes('turbidity')) {
    return 'sediment';
  }
  return 'other';
}

import { useState, useEffect, useCallback } from 'react';
import { hydroApi } from '@/api/hydro';

type Status = 'idle' | 'loading' | 'success' | 'error';

interface UseFetchOptions<T> {
  initialData?: T;
  enabled?: boolean;
  refetchInterval?: number;
  onSuccess?: (data: T) => void;
  onError?: (error: Error) => void;
}

interface UseFetchResult<T> {
  data: T | null;
  error: Error | null;
  status: Status;
  isLoading: boolean;
  isError: boolean;
  isSuccess: boolean;
  refetch: () => Promise<void>;
}

export function useFetch<T>(
  fetchFn: () => Promise<T>,
  options: UseFetchOptions<T> = {}
): UseFetchResult<T> {
  const {
    initialData = null,
    enabled = true,
    refetchInterval,
    onSuccess,
    onError,
  } = options;

  const [data, setData] = useState<T | null>(initialData);
  const [error, setError] = useState<Error | null>(null);
  const [status, setStatus] = useState<Status>('idle');

  const isLoading = status === 'loading';
  const isError = status === 'error';
  const isSuccess = status === 'success';

  const fetchData = useCallback(async () => {
    if (!enabled) return;

    setStatus('loading');
    setError(null);

    try {
      const result = await fetchFn();
      setData(result);
      setStatus('success');
      onSuccess?.(result);
    } catch (err) {
      const error = err instanceof Error ? err : new Error('Unknown error');
      setError(error);
      setStatus('error');
      onError?.(error);
    }
  }, [fetchFn, enabled, onSuccess, onError]);

  useEffect(() => {
    if (enabled) {
      fetchData();
    }
  }, [enabled, fetchData]);

  useEffect(() => {
    if (!refetchInterval) return;

    const intervalId = setInterval(() => {
      if (enabled) {
        fetchData();
      }
    }, refetchInterval);

    return () => clearInterval(intervalId);
  }, [refetchInterval, enabled, fetchData]);

  return {
    data,
    error,
    status,
    isLoading,
    isError,
    isSuccess,
    refetch: fetchData,
  };
}

// Hook simple pour les données hydrologiques
export function useHydroData<T>(
  apiFunction: () => Promise<T>,
  options?: UseFetchOptions<T>
) {
  return useFetch(apiFunction, options);
}

// Exemples d'utilisation :
export const useStations = (options?: UseFetchOptions<Array<{ id: string; name: string }>>) => {
  return useFetch(() => hydroApi.getStations(), options);
};

export const useTimeseries = (params?: { stationId?: string; variable?: string }, options?: UseFetchOptions<Array<{ timestamp: string; value: number }>>) => {
  return useFetch(() => hydroApi.getTimeseries(params || {}), options);
};

// src/contexts/HydroDataContext.tsx
import { createContext, useContext, useEffect, useState, ReactNode } from 'react';
import { getDashboardData } from '@/api/hydro';
import { Station, Catchment, Timeseries, Reservoir } from '@/types/hydro';

interface HydroDataContextType {
  stations: Station[];
  catchments: Catchment[];
  timeseries: Timeseries[];
  reservoirs: Reservoir[];
  loading: boolean;
  error: string | null;
}

const HydroDataContext = createContext<HydroDataContextType>({
  stations: [],
  catchments: [],
  timeseries: [],
  reservoirs: [],
  loading: true,
  error: null,
});

export const HydroDataProvider = ({ children }: { children: ReactNode }) => {
  const [stations, setStations] = useState<Station[]>([]);
  const [catchments, setCatchments] = useState<Catchment[]>([]);
  const [timeseries, setTimeseries] = useState<Timeseries[]>([]);
  const [reservoirs, setReservoirs] = useState<Reservoir[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function fetchData() {
      try {
        const { stations, catchments, timeseries, reservoirs } = await getDashboardData();
        setStations(stations);
        setCatchments(catchments);
        setTimeseries(timeseries);
        setReservoirs(reservoirs);
      } catch (err) {
        setError(err instanceof Error ? err.message : 'Erreur inconnue');
      } finally {
        setLoading(false);
      }
    }

    fetchData();
  }, []);

  return (
    <HydroDataContext.Provider value={{ stations, catchments, timeseries, reservoirs, loading, error }}>
      {children}
    </HydroDataContext.Provider>
  );
};

export const useHydroData = () => useContext(HydroDataContext);
