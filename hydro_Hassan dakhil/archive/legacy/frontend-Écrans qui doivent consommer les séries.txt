import { useState } from 'react';
import { FilterBar } from '../FilterBar';
import { TimeSeriesChart } from '@/components/charts/TimeSeriesChart';
import { DataTable } from '@/components/tables/DataTable';
import type { FilterState } from '@/types/hydro';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
import { BarChart3, Table2, GitCompare, TrendingUp, TrendingDown, Thermometer, CloudRain, Wind } from 'lucide-react';

// Mapping des variables climatiques avec leurs IDs, noms et icônes
const climateVariables = [
  { id: 1, name: 'Température Max', code: 'temp_max', unit: '°C', icon: Thermometer },
  { id: 2, name: 'Température Min', code: 'temp_min', unit: '°C', icon: Thermometer },
  { id: 3, name: 'Température Moyenne', code: 'temp_avg', unit: '°C', icon: Thermometer },
  { id: 4, name: 'Précipitation', code: 'precip', unit: 'mm', icon: CloudRain },
  { id: 5, name: 'Vitesse du Vent', code: 'wind', unit: 'm/s', icon: Wind },
];

// Fonction pour obtenir le nom d'une variable par son ID
const getVariableName = (id: number) => {
  const variable = climateVariables.find(v => v.id === id);
  return variable ? variable.name : `Variable ${id}`;
};

  stations: [],
  variables: [3], // Température Moyenne par défaut
  scenario: 'observed',
  startDate: '1980-01-01',
  endDate: '2025-12-31',
  resolution: 'day',
};

export function ClimateModule() {

            <BarChart3 className="w-4 h-4" />
            Graphique

            <Table2 className="w-4 h-4" />

//src/components/charts/TimeSeriesChart.tsx
import { useMemo } from "react";
import {
  LineChart,
  Line,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  Legend,
  ResponsiveContainer,
  Area,
  ComposedChart,
} from "recharts";
import {
  variables,
  generateMockTimeSeries,
  mockDataGenerators,
  calculateStats,
} from "@/data/mockData";
import type { FilterState } from "@/types/hydro";
import { TrendingUp, TrendingDown, Minus, Download } from "lucide-react";
import { Button } from "@/components/ui/button";

interface TimeSeriesChartProps {
  filters: FilterState;
}

const chartColors = [
  "hsl(200, 80%, 45%)",
  "hsl(185, 80%, 50%)",
  "hsl(160, 65%, 45%)",
  "hsl(35, 85%, 55%)",
  "hsl(280, 65%, 55%)",
  "hsl(340, 75%, 55%)",
];

interface ChartDataPoint {
  date: string;
  [key: string]: string | number;
}

// Définir un type pour les mesures
interface ChartMeasurement {
  date?: string;
  timestamp?: string;
  time?: string;
  value: number;
  [key: string]: unknown;
}

export function TimeSeriesChart({ filters }: TimeSeriesChartProps) {
  const chartData = useMemo(() => {
    if (filters.variables.length === 0)
      return { data: [] as ChartDataPoint[], stats: [], selectedVars: [] };

    const selectedVars = variables.filter((v) =>
      filters.variables.includes(v.id)
    );
    const dataMap = new Map<string, ChartDataPoint>();
    const stats: Array<{
      variable: string;
      unit: string;
      min: number;
      max: number;
      mean: number;
    }> = [];

    selectedVars.forEach((variable) => {
      const generator = mockDataGenerators[variable.code];
      if (!generator) return;

      const measurements = generateMockTimeSeries(
        filters.startDate,
        filters.endDate,
        variable.code,
        generator.base,
        generator.amplitude,
        generator.noise
      );

      const varStats = calculateStats(measurements);
      stats.push({
        variable: variable.name,
        unit: variable.unit,
        min: Math.round(varStats.min * 100) / 100,
        max: Math.round(varStats.max * 100) / 100,
        mean: Math.round(varStats.mean * 100) / 100,
      });

      measurements.forEach((m) => {
        // Convertir la mesure en type connu
        const measurement = m as unknown as ChartMeasurement;

        // Trouver la clé de date - essayer différentes propriétés
        let dateKey = "";
        if (typeof measurement === "object" && measurement !== null) {
          if ("date" in measurement && typeof measurement.date === "string") {
            dateKey = measurement.date;
          } else if (
            "timestamp" in measurement &&
            typeof measurement.timestamp === "string"
          ) {
            dateKey = measurement.timestamp;
          } else if (
            "time" in measurement &&
            typeof measurement.time === "string"
          ) {
            dateKey = measurement.time;
          } else if ("value" in measurement) {
            dateKey = String(measurement.value);
          }
        }

        if (dateKey) {
          const existing = dataMap.get(dateKey) || { date: dateKey };
          existing[variable.code] = measurement.value;
          dataMap.set(dateKey, existing);
        }
      });
    });

    // Sample data for better performance (max 365 points)
    const allData = Array.from(dataMap.values()).sort(
      (a, b) => new Date(a.date).getTime() - new Date(b.date).getTime()
    );

    const step = Math.ceil(allData.length / 365);
    const sampledData = allData.filter((_, i) => i % step === 0);

    return { data: sampledData, stats, selectedVars };
  }, [filters]);

  const exportCSV = () => {
    if (chartData.data.length === 0) return;

    const headers = [
      "Date",
      ...chartData.selectedVars.map((v) => `${v.name} (${v.unit})`),
    ];
    const rows = chartData.data.map((d) => [
      d.date,
      ...chartData.selectedVars.map((v) => d[v.code] ?? ""),
    ]);

    const csv = [headers, ...rows].map((r) => r.join(",")).join("\n");
    const blob = new Blob([csv], { type: "text/csv" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `hydro_data_${filters.startDate}_${filters.endDate}.csv`;
    a.click();
  };

  if (filters.variables.length === 0) {
    return (
      <div className="hydro-card h-full flex items-center justify-center">
        <div className="text-center text-muted-foreground">
          <TrendingUp className="w-12 h-12 mx-auto mb-3 opacity-50" />
          <p>Sélectionnez des variables pour afficher les graphiques</p>
        </div>
      </div>
    );
  }

  return (
    <div className="hydro-card">
      <div className="hydro-card-header">
        <h3 className="font-semibold">Séries Temporelles</h3>
        <Button variant="outline" size="sm" onClick={exportCSV}>
          <Download className="w-4 h-4 mr-2" />
          Export CSV
        </Button>
      </div>

      <div className="hydro-card-body">
        {/* Stats Cards */}
        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-6 gap-3 mb-6">
          {chartData.stats.map((stat, i) => (
            <div key={stat.variable} className="stat-card">
              <div
                className="text-xs text-muted-foreground mb-1 truncate"
                title={stat.variable}
              >
                {stat.variable}
              </div>
              <div className="space-y-1">
                <div className="flex items-center justify-between text-xs">
                  <span className="text-muted-foreground">Min</span>
                  <span className="font-mono font-medium text-blue-600">
                    {stat.min} {stat.unit}
                  </span>
                </div>
                <div className="flex items-center justify-between text-xs">
                  <span className="text-muted-foreground">Max</span>
                  <span className="font-mono font-medium text-red-600">
                    {stat.max} {stat.unit}
                  </span>
                </div>
                <div className="flex items-center justify-between text-xs">
                  <span className="text-muted-foreground">Moy</span>
                  <span className="font-mono font-medium text-foreground">
                    {stat.mean} {stat.unit}
                  </span>
                </div>
              </div>
            </div>
          ))}
        </div>

        {/* Chart */}
        <div className="chart-container">
          <ResponsiveContainer width="100%" height="100%">
            <ComposedChart
              data={chartData.data}
              margin={{ top: 10, right: 10, left: 0, bottom: 0 }}
            >
              <defs>
                {chartData.selectedVars.map((v, i) => (
                  <linearGradient
                    key={v.code}
                    id={`gradient-${v.code}`}
                    x1="0"
                    y1="0"
                    x2="0"
                    y2="1"
                  >
                    <stop
                      offset="5%"
                      stopColor={chartColors[i % chartColors.length]}
                      stopOpacity={0.3}
                    />
                    <stop
                      offset="95%"
                      stopColor={chartColors[i % chartColors.length]}
                      stopOpacity={0}
                    />
                  </linearGradient>
                ))}
              </defs>
              <CartesianGrid
                strokeDasharray="3 3"
                stroke="hsl(var(--border))"
                opacity={0.5}
              />
              <XAxis
                dataKey="date"
                tick={{ fontSize: 11 }}
                tickFormatter={(value) => {
                  const date = new Date(value);
                  return `${date.getMonth() + 1}/${date
                    .getFullYear()
                    .toString()
                    .slice(2)}`;
                }}
                stroke="hsl(var(--muted-foreground))"
              />
              <YAxis
                tick={{ fontSize: 11 }}
                stroke="hsl(var(--muted-foreground))"
              />
              <Tooltip
                contentStyle={{
                  backgroundColor: "hsl(var(--card))",
                  border: "1px solid hsl(var(--border))",
                  borderRadius: "8px",
                  fontSize: "12px",
                }}
                labelFormatter={(label) =>
                  new Date(label).toLocaleDateString("fr-FR")
                }
              />
              <Legend
                wrapperStyle={{ fontSize: "12px" }}
                formatter={(value) => {
                  const variable = chartData.selectedVars.find(
                    (v) => v.code === value
                  );
                  return variable
                    ? `${variable.name} (${variable.unit})`
                    : value;
                }}
              />
              {chartData.selectedVars.map((v, i) => (
                <Area
                  key={v.code}
                  type="monotone"
                  dataKey={v.code}
                  stroke={chartColors[i % chartColors.length]}
                  fill={`url(#gradient-${v.code})`}
                  strokeWidth={2}
                  dot={false}
                  activeDot={{ r: 4, strokeWidth: 2 }}
                />
              ))}
            </ComposedChart>
          </ResponsiveContainer>
        </div>
      </div>
    </div>
  );
}

//src/components/tables/DataTable.tsx
import { useState, useMemo } from 'react';
import { ChevronUp, ChevronDown, Download, Search, ChevronLeft, ChevronRight } from 'lucide-react';
import { variables, generateMockTimeSeries, mockDataGenerators } from '@/data/mockData';
import type { FilterState } from '@/types/hydro';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { cn } from '@/lib/utils';

interface DataTableProps {
  filters: FilterState;
}

const ROWS_PER_PAGE = 15;

// Définir le type de mesure basé sur ce que retourne generateMockTimeSeries
interface TableMeasurement {
  date?: string;
  timestamp?: string;
  time?: string;
  value: number;
  [key: string]: unknown; // Signature d'index pour flexibilité
}

export function DataTable({ filters }: DataTableProps) {
  const [sortColumn, setSortColumn] = useState<string>('date');
  const [sortDirection, setSortDirection] = useState<'asc' | 'desc'>('desc');
  const [searchTerm, setSearchTerm] = useState('');
  const [currentPage, setCurrentPage] = useState(1);

  const tableData = useMemo(() => {
    if (filters.variables.length === 0) return { rows: [], columns: [] };

    const selectedVars = variables.filter(v => filters.variables.includes(v.id));
    const columns = [
      { key: 'date', label: 'Date' },
      ...selectedVars.map(v => ({ key: v.code, label: `${v.name} (${v.unit})` })),
    ];

    const dataMap = new Map<string, Record<string, string | number>>();

    selectedVars.forEach((variable) => {
      const generator = mockDataGenerators[variable.code];
      if (!generator) return;

      const measurements = generateMockTimeSeries(
        filters.startDate,
        filters.endDate,
        variable.code,
        generator.base,
        generator.amplitude,
        generator.noise
      );

      measurements.forEach((m) => {
        // Convertir en unknown puis en TableMeasurement pour éviter les erreurs de type
        const measurement = m as unknown as TableMeasurement;
        
        // Extraire la date - essayer différentes propriétés possibles
        let dateKey = '';
        if (typeof measurement === 'object' && measurement !== null) {
          if ('date' in measurement && typeof measurement.date === 'string') {
            dateKey = measurement.date;
          } else if ('timestamp' in measurement && typeof measurement.timestamp === 'string') {
            dateKey = measurement.timestamp;
          } else if ('time' in measurement && typeof measurement.time === 'string') {
            dateKey = measurement.time;
          } else if ('value' in measurement) {
            dateKey = String(measurement.value);
          }
        }
        
        if (dateKey) {
          const existing = dataMap.get(dateKey) || { date: dateKey };
          existing[variable.code] = measurement.value;
          dataMap.set(dateKey, existing);
        }
      });
    });

    const rows = Array.from(dataMap.values());
    return { rows, columns };
  }, [filters]);

  const sortedAndFilteredData = useMemo(() => {
    let data = [...tableData.rows];

    // Filter
    if (searchTerm) {
      data = data.filter(row => 
        Object.values(row).some(val => 
          String(val).toLowerCase().includes(searchTerm.toLowerCase())
        )
      );
    }

    // Sort
    data.sort((a, b) => {
      const aVal = a[sortColumn];
      const bVal = b[sortColumn];
      
      if (sortColumn === 'date') {
        const dateA = new Date(aVal as string).getTime();
        const dateB = new Date(bVal as string).getTime();
        return sortDirection === 'asc' ? dateA - dateB : dateB - dateA;
      }
      
      const numA = Number(aVal) || 0;
      const numB = Number(bVal) || 0;
      return sortDirection === 'asc' ? numA - numB : numB - numA;
    });

    return data;
  }, [tableData.rows, sortColumn, sortDirection, searchTerm]);

  const paginatedData = useMemo(() => {
    const start = (currentPage - 1) * ROWS_PER_PAGE;
    return sortedAndFilteredData.slice(start, start + ROWS_PER_PAGE);
  }, [sortedAndFilteredData, currentPage]);

  const totalPages = Math.ceil(sortedAndFilteredData.length / ROWS_PER_PAGE);

  const handleSort = (column: string) => {
    if (sortColumn === column) {
      setSortDirection(prev => prev === 'asc' ? 'desc' : 'asc');
    } else {
      setSortColumn(column);
      setSortDirection('desc');
    }
  };

  const exportCSV = () => {
    const headers = tableData.columns.map(c => c.label);
    const rows = sortedAndFilteredData.map(row => 
      tableData.columns.map(c => row[c.key] ?? '')
    );
    
    const csv = [headers, ...rows].map(r => r.join(',')).join('\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `hydro_table_${filters.startDate}_${filters.endDate}.csv`;
    a.click();
  };

  if (filters.variables.length === 0) {
    return (
      <div className="hydro-card h-full flex items-center justify-center">
        <div className="text-center text-muted-foreground">
          <p>Sélectionnez des variables pour afficher les données</p>
        </div>
      </div>
    );
  }

  return (
    <div className="hydro-card">
      <div className="hydro-card-header">
        <h3 className="font-semibold">Données Tabulaires</h3>
        <div className="flex items-center gap-3">
          <div className="relative">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
            <Input
              placeholder="Rechercher..."
              value={searchTerm}
              onChange={(e) => {
                setSearchTerm(e.target.value);
                setCurrentPage(1);
              }}
              className="pl-9 w-48"
            />
          </div>
          <Button variant="outline" size="sm" onClick={exportCSV}>
            <Download className="w-4 h-4 mr-2" />
            Export CSV
          </Button>
        </div>
      </div>

      <div className="overflow-x-auto">
        <table className="data-table">
          <thead>
            <tr>
              {tableData.columns.map((column) => (
                <th
                  key={column.key}
                  onClick={() => handleSort(column.key)}
                  className="cursor-pointer hover:bg-muted/70 transition-colors"
                >
                  <div className="flex items-center gap-2">
                    <span className="truncate">{column.label}</span>
                    <div className="flex flex-col">
                      <ChevronUp 
                        className={cn(
                          'w-3 h-3 -mb-1',
                          sortColumn === column.key && sortDirection === 'asc' 
                            ? 'text-primary' 
                            : 'text-muted-foreground/40'
                        )} 
                      />
                      <ChevronDown 
                        className={cn(
                          'w-3 h-3 -mt-1',
                          sortColumn === column.key && sortDirection === 'desc' 
                            ? 'text-primary' 
                            : 'text-muted-foreground/40'
                        )} 
                      />
                    </div>
                  </div>
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {paginatedData.map((row, i) => (
              <tr key={i}>
                {tableData.columns.map((column) => (
                  <td key={column.key} className="font-mono text-sm">
                    {column.key === 'date' 
                      ? new Date(row[column.key] as string).toLocaleDateString('fr-FR')
                      : typeof row[column.key] === 'number' 
                        ? (row[column.key] as number).toFixed(2)
                        : row[column.key] ?? '-'
                    }
                  </td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {/* Pagination */}
      <div className="px-5 py-4 border-t border-border flex items-center justify-between">
        <div className="text-sm text-muted-foreground">
          {sortedAndFilteredData.length} enregistrements • Page {currentPage} / {totalPages}
        </div>
        <div className="flex items-center gap-2">
          <Button
            variant="outline"
            size="sm"
            onClick={() => setCurrentPage(p => Math.max(1, p - 1))}
            disabled={currentPage === 1}
          >
            <ChevronLeft className="w-4 h-4" />
          </Button>
          <Button
            variant="outline"
            size="sm"
            onClick={() => setCurrentPage(p => Math.min(totalPages, p + 1))}
            disabled={currentPage === totalPages}
          >
            <ChevronRight className="w-4 h-4" />
          </Button>
        </div>
      </div>
    </div>
  );
}
